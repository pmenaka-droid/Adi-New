def get_db():
   """Open one database connection per request"""
   db = sqlite3.connect(DATABASE)
   db.row_factory = sqlite3.Row
   db.execute("PRAGMA foreign_keys = ON")
   return db


def init_db():
   """Create all 5 tables when the app starts"""
   db = get_db()
   db.executescript('''
      CREATE TABLE IF NOT EXISTS users (
            user_id    INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT NOT NULL UNIQUE,
            email      TEXT NOT NULL UNIQUE,
            password TEXT NOT NULL
      );


      CREATE TABLE IF NOT EXISTS recipes (
            recipe_id      INTEGER PRIMARY KEY AUTOINCREMENT,
            title          TEXT NOT NULL,
            instructions TEXT NOT NULL,
            category       TEXT,
            cook_time      INTEGER,
            difficulty     TEXT,
            dietary        TEXT,
            image_file     TEXT,
            likes          INTEGER DEFAULT 0,
            dislikes       INTEGER DEFAULT 0,
            uploader_id    INTEGER,
               FOREIGN KEY (uploader_id) REFERENCES users(user_id)
          );


          CREATE TABLE IF NOT EXISTS ingredients (
               ingredient_id INTEGER PRIMARY KEY AUTOINCREMENT,
               name            TEXT NOT NULL UNIQUE
          );


          CREATE TABLE IF NOT EXISTS recipe_ingredients (
               recipe_id       INTEGER,
               ingredient_id INTEGER,
               quantity        TEXT,
               PRIMARY KEY (recipe_id, ingredient_id),
               FOREIGN KEY (recipe_id)
                  REFERENCES recipes(recipe_id),
               FOREIGN KEY (ingredient_id)
                  REFERENCES ingredients(ingredient_id)
          );


          CREATE TABLE IF NOT EXISTS saved_recipes (
               user_id     INTEGER,
               recipe_id INTEGER,
               PRIMARY KEY (user_id, recipe_id),
               FOREIGN KEY (user_id)
                  REFERENCES users(user_id),
               FOREIGN KEY (recipe_id)
                  REFERENCES recipes(recipe_id)
          );
   ''')
   db.commit()
   db.close()

@app.route('/')
def index():
   db = get_db()


   ingredient_query = request.args.get('ingredient', '').strip()
   category_query = request.args.get('category', '').strip()
   difficulty_query = request.args.get('difficulty', '').strip()
   dietary_query = request.args.get('dietary', '').strip()
   cook_time_query = request.args.get('cook_time', '').strip()


   sql = '''
         SELECT DISTINCT recipes.*
         FROM recipes
         LEFT JOIN recipe_ingredients ON recipes.recipe_id =
recipe_ingredients.recipe_id
         LEFT JOIN ingredients ON recipe_ingredients.ingredient_id =
ingredients.ingredient_id
         WHERE 1=1
   '''
   params = []


   if ingredient_query:
         sql += ' AND ingredients.name LIKE ?'
         params.append('%' + ingredient_query + '%')


   if category_query:
         sql += ' AND recipes.category = ?'
         params.append(category_query)


   if difficulty_query:
         sql += ' AND recipes.difficulty = ?'
         params.append(difficulty_query)
   if dietary_query:
       sql += ' AND recipes.dietary = ?'
       params.append(dietary_query)


   if cook_time_query == 'under30':
       sql += ' AND recipes.cook_time < 30'
   elif cook_time_query == '30to60':
       sql += ' AND recipes.cook_time >= 30 AND recipes.cook_time <= 60'
   elif cook_time_query == 'over60':
       sql += ' AND recipes.cook_time > 60'


   recipes = db.execute(sql, params).fetchall()


   categories = [row['category'] for row in db.execute('SELECT DISTINCT category FROM
recipes WHERE category IS NOT NULL').fetchall()]
   difficulties = [row['difficulty'] for row in db.execute('SELECT DISTINCT difficulty
FROM recipes WHERE difficulty IS NOT NULL').fetchall()]
   dietary_options = [row['dietary'] for row in db.execute('SELECT DISTINCT dietary
FROM recipes WHERE dietary IS NOT NULL').fetchall()]


   db.close()
   return render_template(
       'index.html',
       recipes=recipes,
       categories=categories,
       difficulties=difficulties,
       dietary_options=dietary_options
   )




@app.route('/register', methods=['GET', 'POST'])
def register():
   if request.method == 'POST':
       username = request.form['username']
       email = request.form['email']
       password = request.form['password']
       hashed_pw = generate_password_hash(password)


       db = get_db()
       try:
              db.execute(
                 'INSERT INTO users (username, email, password) VALUES (?, ?, ?)',
                (username, email, hashed_pw)
           )
           db.commit()
      except sqlite3.IntegrityError:
           flash('Username or email already taken.')
           db.close()
           return redirect(url_for('register'))
      db.close()
      return redirect(url_for('login'))


   return render_template('register.html')




@app.route('/login', methods=['GET', 'POST'])
def login():
   if request.method == 'POST':
      email = request.form['email']
      password = request.form['password']


      db = get_db()
      row = db.execute('SELECT * FROM users WHERE email = ?', (email,)).fetchone()
      db.close()


      if row and check_password_hash(row['password'], password):
           user = User(row['user_id'], row['username'], row['email'])
           login_user(user)
           return redirect(url_for('index'))
      else:
           flash('Invalid email or password.')


   return render_template('login.html')

@app.route('/save/<int:recipe_id>', methods=['POST'])
@login_required
def toggle_save(recipe_id):
   db = get_db()


   already_saved = db.execute(
      'SELECT * FROM saved_recipes WHERE user_id = ? AND recipe_id = ?',
      (current_user.id, recipe_id)
   ).fetchone()


   if already_saved:
      db.execute(
           'DELETE FROM saved_recipes WHERE user_id = ? AND recipe_id = ?',
           (current_user.id, recipe_id)
      )
   else:
      db.execute(
           'INSERT INTO saved_recipes (user_id, recipe_id) VALUES (?, ?)',
           (current_user.id, recipe_id)
      )


   db.commit()
   db.close()
   return redirect(url_for('recipe', recipe_id=recipe_id))

COOK_TIME_BY_DIFFICULTY = {
    'Easy': 20,
    'Medium': 40,
    'High': 75,
}

def estimate_cook_times():
    db = get_db()


    missing = db.execute(
       'SELECT recipe_id, difficulty FROM recipes WHERE cook_time IS NULL'
    ).fetchall()


    updated_counts = {'Easy': 0, 'Medium': 0, 'High': 0, 'Unknown': 0}


    for recipe in missing:
       difficulty = recipe['difficulty']
       estimated_time = COOK_TIME_BY_DIFFICULTY.get(difficulty)


       if estimated_time is None:
            # Difficulty itself was somehow missing too — fall back to a
            # safe default rather than leaving cook_time blank again.
            estimated_time = 30
            updated_counts['Unknown'] += 1
       else:
            updated_counts[difficulty] += 1


       db.execute(
            'UPDATE recipes SET cook_time = ? WHERE recipe_id = ?',
            (estimated_time, recipe['recipe_id'])
       )
   db.commit()
   db.close()


   print(f"Estimated cook_time for {len(missing)} recipes that had none:")
   for key, count in updated_counts.items():
      if count:
          print(f"   {key}: {count}")

   estimate_cook_times()

def parse_ingredient_list(raw_value):
  try:
         ingredients = ast.literal_eval(raw_value)
         if isinstance(ingredients, list):
                return [i.strip() for i in ingredients if i and i.strip()]
  except (ValueError, SyntaxError):
         pass
  return []


def recipe_already_imported(db, title):
  row = db.execute('SELECT recipe_id FROM recipes WHERE title = ?',
(title,)).fetchone()
  return row is not None


def import_recipes():
  db = get_db()
  total_added = 0
  total_skipped = 0


  with open(CSV_FILE, newline='', encoding='utf-8') as f:
         reader = csv.DictReader(f)


         for row in reader:
                if total_added >= MAX_RECIPES_TO_IMPORT:
                   break


                title = (row.get('Title') or '').strip()
            instructions = (row.get('Instructions') or '').strip()
            raw_ingredients = row.get('Ingredients') or '[]'


            if not title or not instructions:
                total_skipped += 1
                continue


            if recipe_already_imported(db, title):
                total_skipped += 1
                continue


            ingredient_list = parse_ingredient_list(raw_ingredients)
            if not ingredient_list:
                total_skipped += 1
                continue


            cursor = db.execute(
                '''INSERT INTO recipes (title, instructions, category, cook_time,
difficulty)
                    VALUES (?, ?, ?, ?, ?)''',
                (title, instructions, 'Uncategorized', None, 'Medium')
            )
            recipe_id = cursor.lastrowid


            for ingredient_phrase in ingredient_list:
                name = ingredient_phrase.lower()
                db.execute(
                     'INSERT OR IGNORE INTO ingredients (name) VALUES (?)',
                     (name,)
                )
                ingredient_row = db.execute(
                     'SELECT ingredient_id FROM ingredients WHERE name = ?',
                     (name,)
                ).fetchone()


                db.execute(
    '''INSERT OR IGNORE INTO recipe_ingredients (recipe_id, ingredient_id, quantity)
      VALUES (?, ?, ?)''',
    (recipe_id, ingredient_row['ingredient_id'], '')
)


            total_added += 1
 db.commit()
 db.close()
 print(f"Done. Added {total_added} recipes ({total_skipped} skipped).")

def find_image_file(image_name):
  if not image_name or image_name.strip() == '' or image_name.strip() == '#NAME?':
      return None


  name = image_name.strip()
  search_folders = [
      IMAGE_FOLDER,
      os.path.join(IMAGE_FOLDER, IMAGE_FOLDER),
  ]


  for folder in search_folders:
      for ext in VALID_EXTENSIONS:
             candidate = os.path.join(folder, name + ext)
             if os.path.isfile(candidate):
                return candidate


  return None

def get_db():
    db = sqlite3.connect(DATABASE)
    db.row_factory = sqlite3.Row
    db.execute("PRAGMA foreign_keys = ON")
    return db


def guess_category(title):
    title_lower = title.lower()
    for category, keywords in CATEGORY_KEYWORDS:
       for keyword in keywords:
               if keyword in title_lower:
                   return category
    return 'Other'


def guess_difficulty(ingredient_count, instruction_word_count):
    if ingredient_count <= 5 and instruction_word_count <= 80:
       return 'Easy'
    if ingredient_count >= 12 or instruction_word_count >= 250:
       return 'High'
    return 'Medium'


def reassign():
   db = get_db()


   uncategorized = db.execute(
      "SELECT recipe_id, title FROM recipes WHERE category = 'Uncategorized'"
   ).fetchall()


   category_counts = {}
   for recipe in uncategorized:
      new_category = guess_category(recipe['title'])
      db.execute(
             'UPDATE recipes SET category = ? WHERE recipe_id = ?',
             (new_category, recipe['recipe_id'])
      )
      category_counts[new_category] = category_counts.get(new_category, 0) + 1


   all_recipes = db.execute('SELECT recipe_id, instructions FROM recipes').fetchall()


   difficulty_counts = {'Easy': 0, 'Medium': 0, 'High': 0}
   for recipe in all_recipes:
      ingredient_count = db.execute(
             'SELECT COUNT(*) as c FROM recipe_ingredients WHERE recipe_id = ?',
             (recipe['recipe_id'],)
      ).fetchone()['c']


      instruction_word_count = len((recipe['instructions'] or '').split())


      new_difficulty = guess_difficulty(ingredient_count, instruction_word_count)
      db.execute(
             'UPDATE recipes SET difficulty = ? WHERE recipe_id = ?',
             (new_difficulty, recipe['recipe_id'])
      )
      difficulty_counts[new_difficulty] += 1


   db.commit()
   db.close()


   print(f"Categories reassigned for {len(uncategorized)} previously 'Uncategorized'
recipes:")
   for cat, count in sorted(category_counts.items(), key=lambda x: -x[1]):
      print(f"     {cat}: {count}")


   print(f"\nDifficulty recomputed for all {len(all_recipes)} recipes:")
   for diff, count in difficulty_counts.items():
      print(f"   {diff}: {count}")
